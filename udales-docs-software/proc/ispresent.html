    logical function ispresent(str)
      implicit none
      character(len=*), intent(in) :: str
      integer :: pos

      pos = index(slicevars, str)   ! returns the position of the substring str in slicevars. If not found, it returns 0.
      ispresent = (pos > 0)
    end function ispresent
